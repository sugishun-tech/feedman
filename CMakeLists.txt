cmake_minimum_required(VERSION 3.16)
project(feedman VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(FEEDMAN_GUI "Build GTK3 desktop UI" ON)
option(FEEDMAN_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(FEEDMAN_NATIVE "Optimize specifically for this CPU (do not redistribute binaries)" OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(CORE REQUIRED IMPORTED_TARGET libcurl>=7.62 libxml-2.0>=2.9 sqlite3>=3.24 json-c>=0.13)
add_library(feedman_core STATIC src/util.c src/config.c src/parser.c src/storage.c src/fetch.c src/import.c)
target_include_directories(feedman_core PUBLIC include)
target_link_libraries(feedman_core PUBLIC PkgConfig::CORE Threads::Threads)
target_compile_definitions(feedman_core PRIVATE _FILE_OFFSET_BITS=64)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Wformat=2 -Wshadow -Wstrict-prototypes)
  target_compile_options(feedman_core PRIVATE -Wall -Wextra -Wpedantic -Wformat=2 -Wshadow -Wstrict-prototypes)
  if(FEEDMAN_NATIVE)
    add_compile_options(-march=native)
    target_compile_options(feedman_core PRIVATE -march=native)
  endif()
  if(FEEDMAN_SANITIZE)
    target_compile_options(feedman_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(feedman_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(NOT FEEDMAN_SANITIZE)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
  if(ipo_supported)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    set_property(TARGET feedman_core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  endif()
endif()
add_executable(feedman src/main.c)
target_link_libraries(feedman PRIVATE feedman_core)
if(FEEDMAN_GUI)
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24)
  target_sources(feedman PRIVATE src/ui.c src/reader.c)
  target_compile_definitions(feedman PRIVATE FEEDMAN_WITH_GUI=1)
  target_link_libraries(feedman PRIVATE PkgConfig::GTK)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(feedman_tests tests/test_core.c tests/test_server.c)
  target_include_directories(feedman_tests PRIVATE tests)
  target_link_libraries(feedman_tests PRIVATE feedman_core)
  target_compile_definitions(feedman_tests PRIVATE FM_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME core COMMAND feedman_tests)
  set_tests_properties(core PROPERTIES TIMEOUT 60)
  if(FEEDMAN_GUI)
    add_executable(feedman_ui_tests tests/test_ui.c src/ui.c src/reader.c)
    target_compile_definitions(feedman_ui_tests PRIVATE FEEDMAN_WITH_GUI=1 FEEDMAN_UI_TESTING=1)
    target_link_libraries(feedman_ui_tests PRIVATE feedman_core PkgConfig::GTK)
    find_program(XVFB_RUN xvfb-run)
    if(XVFB_RUN)
      add_test(NAME gui COMMAND ${XVFB_RUN} -a -s "-screen 0 1600x1000x24" $<TARGET_FILE:feedman_ui_tests>)
      set_tests_properties(gui PROPERTIES TIMEOUT 45 ENVIRONMENT "NO_AT_BRIDGE=1;G_DEBUG=fatal-criticals")
    endif()
  endif()
  add_test(NAME cli_help COMMAND feedman --help)
  add_test(NAME cli_version COMMAND feedman --version)
  add_test(NAME cli_bad_date COMMAND feedman --date 20260230 --check-feeds)
  set_tests_properties(cli_bad_date PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli_bad_threads COMMAND feedman --threads 0 --check-feeds)
  set_tests_properties(cli_bad_threads PROPERTIES WILL_FAIL TRUE)
endif()
include(GNUInstallDirs)
install(TARGETS feedman RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES feeds.example.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/feedman)
if(FEEDMAN_GUI)
  install(FILES packaging/feedman.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()
